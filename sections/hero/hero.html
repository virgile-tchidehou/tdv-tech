// SMART-SOJA — FreeRTOS Architecture
#include <Arduino.h>
#include <FreeRTOS.h>

QueueHandle_t     sensorQueue;
SemaphoreHandle_t serialMutex;

void task_sensors(void *pv) {
  for (;;) {
    SensorData d = readAllSensors();
    xQueueSend(sensorQueue, &d, 0);
    vTaskDelay(pdMS_TO_TICKS(500));
  }
}

void setup() {
  sensorQueue = xQueueCreate(10,
    sizeof(SensorData));
  xTaskCreatePinnedToCore(
    task_sensors, "sensors",
    4096, NULL, 2, NULL, 1
  );
}
